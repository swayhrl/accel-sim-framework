1268: void baseline_cache::invalidate() {
1269:   // The pre-existing kernel-end flush can be requested after an SM has no
1270:   // runnable threads but before its conventional L1 misses have drained.
1271:   // Invalidating such a tag lets a later MSHR merge reserve a second tag while
1272:   // the original fill owner still points at the first cache index.  The final
1273:   // fill then retires the owner/MSHR but leaves that second tag RESERVED.
1274:   // Preserve the request and perform the same invalidation once the accepted
1275:   // miss lifecycle has drained; no request, response, or DTC accounting path
1276:   // is changed.
1277:   m_invalidate_pending = true;
1278:   retire_pending_invalidate();
1279: }

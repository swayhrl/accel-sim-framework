import os,stat,json,time,pathlib,subprocess
from datetime import datetime,timezone
OUT=pathlib.Path('/workspace/worktrees/framework-s-owner-growth-20261019-v2/docs/ep_l2/operations/full_system_20261019_v2/source_S_owner_growth')
KNOWN=['/workspace/fast64-repaired-ramp-closeout-supervisor.log','/workspace/fast64-3-v2-terminal-aggregation-supervisor.log','/workspace/c16_assets/c16-a/background/logs/c16_background_asset_supervisor.log','/workspace/c16_assets/c16-a/background/status/C16_A_BACKGROUND_STATUS.json']
def metadata(p):
 s=os.stat(p);return dict(device=s.st_dev,inode=s.st_ino,size=s.st_size,allocated_bytes=s.st_blocks*512,nlink=s.st_nlink,mtime_ns=s.st_mtime_ns)
def link(p):
 try:return os.readlink(p)
 except OSError:return None
def sample():
 start=time.monotonic();r=dict(utc=datetime.now(timezone.utc).isoformat(),monotonic=start,self_pid=os.getpid(),mount_namespace=link('/proc/self/ns/mnt'),filesystems={},known_paths={},processes=[],errors={},fd_limit=512,time_limit_seconds=12)
 for p in ['/workspace','/root/data','/root/share']:
  v=os.statvfs(p);r['filesystems'][p]=dict(device=os.stat(p).st_dev,fsid=v.f_fsid,available_bytes=v.f_bavail*v.f_frsize,free_bytes=v.f_bfree*v.f_frsize,total_bytes=v.f_blocks*v.f_frsize,available_inodes=v.f_favail)
 for p in KNOWN:
  try:r['known_paths'][p]=metadata(p)
  except OSError as e:r['known_paths'][p]={'error':type(e).__name__}
 for d in pathlib.Path('/proc').iterdir():
  if not d.name.isdigit():continue
  if time.monotonic()-start>12:r['errors']['time_limit_reached']=True;break
  try:
   uid=d.stat().st_uid
   row=dict(pid=int(d.name),uid=uid,comm=(d/'comm').read_text().strip(),start_ticks=int((d/'stat').read_text().rsplit(')',1)[1].split()[19]),io={})
   row['io']={k:int(v) for k,v in (x.split(':') for x in (d/'io').read_text().splitlines())}
   row.update(cwd=link(d/'cwd'),exe=link(d/'exe'),mount_namespace=link(d/'ns/mnt'),write_fds=[],deleted_regular_fds=[])
   if uid==os.getuid():
    fds=list((d/'fd').iterdir());row['fd_count']=len(fds);row['fd_truncated']=len(fds)>512
    for fd in fds[:512]:
     try:
      s=fd.stat()
      if not stat.S_ISREG(s.st_mode):continue
      flags=int(next(x.split(':',1)[1].strip() for x in (d/'fdinfo'/fd.name).read_text().splitlines() if x.startswith('flags:')),8)
      if not (flags&3) and s.st_nlink:continue
      target=link(fd);m=metadata(fd);m.update(fd=int(fd.name),path=target,access_mode=flags&3)
      if flags&3:row['write_fds'].append(m)
      if s.st_nlink==0:row['deleted_regular_fds'].append(m)
     except (OSError,ValueError,StopIteration):pass
   r['processes'].append(row)
  except (OSError,ValueError) as e:r['errors'][type(e).__name__]=r['errors'].get(type(e).__name__,0)+1
 r['duration_seconds']=time.monotonic()-start
 return r
for i in range(2):
 r=sample();(OUT/f'GROWTH_SNAPSHOT_{i+1}.json').write_text(json.dumps(r,indent=2)+'\n')
 print(json.dumps(dict(snapshot=i+1,utc=r['utc'],available_bytes=r['filesystems']['/workspace']['available_bytes'],process_count=len(r['processes']),duration_seconds=r['duration_seconds'],errors=r['errors'])),flush=True)
 if i==0:time.sleep(30)
